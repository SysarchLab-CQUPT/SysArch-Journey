先约定这颗教学数据通路的边界

32 位、顺序单发射、五级流水；仅支持 ADD / SUB / ADDI / LW / SW / BEQ。指令与数据分开存储，组合读、上升沿写，BEQ 在 EX 级决策。这个固定延迟模型用于解释机制；接同步 BRAM、Cache 或 AXI 时必须重新设计握手和等待。

三条阅读线索
先找寄存器边界。以 ID/EX 为例，当前指令的 PC、源寄存器号、读值、立即数和后级控制一起进入下一拍。非阻塞赋值的右值来自边沿前，所以多级寄存器同时更新仍然只前进一级。
再问数据何时可用。ALU 的结果在 EX 结束时可用，下一拍可以从 EX/MEM 送回 EX；load 在 EX 得到的只是地址，数据到 MEM 结束才可用。相邻消费者必须留在 ID 一拍。
最后看副作用门控。暂停保持 PC 和 IF/ID，却把 ID/EX 置为无效；分支冲刷取消年轻指令；寄存器写使能和存储器写使能都与 valid 相与。看到无效级仍有旧地址不意味着真的写了内存。

pipeline_demo.v
`timescale 1ns/1ps
// Five-stage teaching datapath, fixed-latency Harvard memory interfaces.
// Instruction and data reads are COMBINATIONAL; data writes use posedge clk.
module pipeline_demo (
    input wire clk, rst,
    output reg [31:0] imem_addr,
    input wire [31:0] imem_rdata,
    output wire [31:0] dmem_addr, dmem_wdata,
    output wire dmem_read, dmem_write,
    input wire [31:0] dmem_rdata,
    output wire commit_valid, commit_we,
    output wire [31:0] commit_pc, commit_data,
    output wire [4:0] commit_rd
);
    // IF/ID: the instruction and ITS OWN address travel together.
    reg ifid_valid;
    reg [31:0] ifid_pc, ifid_instr;
    wire [4:0] d_rs1, d_rs2, d_rd;
    wire [31:0] d_imm, d_rv1, d_rv2;
    wire d_use1, d_use2, d_rw, d_mr, d_mw, d_ai, d_sub, d_br, d_legal;
    decoder dec (
        .instr(ifid_instr), .rs1(d_rs1), .rs2(d_rs2), .rd(d_rd),
        .imm(d_imm), .uses_rs1(d_use1), .uses_rs2(d_use2),
        .reg_write(d_rw), .mem_read(d_mr), .mem_write(d_mw),
        .alu_imm(d_ai), .alu_sub(d_sub), .branch(d_br), .legal(d_legal)
    );

    // ID/EX contains operands, register numbers and future-stage controls.
    reg idex_valid;
    reg [31:0] idex_pc, idex_rv1, idex_rv2, idex_imm;
    reg [4:0] idex_rs1, idex_rs2, idex_rd;
    reg idex_use1, idex_use2, idex_rw, idex_mr, idex_mw;
    reg idex_ai, idex_sub, idex_br;
    // EX/MEM holds an address/ALU result and forwarded store data.
    reg exmem_valid;
    reg [31:0] exmem_pc, exmem_result, exmem_store;
    reg [4:0] exmem_rd;
    reg exmem_rw, exmem_mr, exmem_mw;
    // MEM/WB holds the load result separately from the ALU result.
    reg memwb_valid;
    reg [31:0] memwb_pc, memwb_result, memwb_load;
    reg [4:0] memwb_rd;
    reg memwb_rw, memwb_mr;
    wire [31:0] wb_data = memwb_mr ? memwb_load : memwb_result;
    wire wb_we = !rst && memwb_valid && memwb_rw && memwb_rd != 0;
    regfile rf (
        .clk(clk), .rst(rst), .rs1(d_rs1), .rs2(d_rs2),
        .rd(memwb_rd), .we(wb_we), .wd(wb_data), .rv1(d_rv1), .rv2(d_rv2)
    );

    wire [1:0] forward_a, forward_b;
    forward_unit fwd (
        .ex_valid(idex_valid), .uses_rs1(idex_use1), .uses_rs2(idex_use2),
        .rs1(idex_rs1), .rs2(idex_rs2),
        .m_valid(exmem_valid), .m_reg_write(exmem_rw),
        .m_mem_read(exmem_mr), .m_rd(exmem_rd),
        .w_valid(memwb_valid), .w_reg_write(memwb_rw), .w_rd(memwb_rd),
        .forward_a(forward_a), .forward_b(forward_b)
    );
    wire [31:0] ex_a = (forward_a == 2'b10) ? exmem_result :
                       (forward_a == 2'b01) ? wb_data : idex_rv1;
    wire [31:0] ex_rs2 = (forward_b == 2'b10) ? exmem_result :
                         (forward_b == 2'b01) ? wb_data : idex_rv2;
    // Store/branch use ex_rs2; ALU's second operand may instead be immediate.
    wire [31:0] ex_b = idex_ai ? idex_imm : ex_rs2;
    wire [31:0] ex_result = idex_sub ? ex_a - ex_b : ex_a + ex_b;
    wire redirect = idex_valid && idex_br && ex_a == ex_rs2;
    wire [31:0] redirect_pc = idex_pc + idex_imm;
    wire load_use, pc_en, ifid_en, ifid_flush, idex_flush;
    hazard_unit hazards (
        .d_valid(ifid_valid && d_legal), .d_uses_rs1(d_use1),
        .d_uses_rs2(d_use2), .d_rs1(d_rs1), .d_rs2(d_rs2),
        .e_valid(idex_valid), .e_mem_read(idex_mr), .e_rd(idex_rd),
        .redirect(redirect), .load_use(load_use), .pc_en(pc_en),
        .ifid_en(ifid_en), .ifid_flush(ifid_flush), .idex_flush(idex_flush)
    );

    // IF: synchronous reset > redirect > hold/advance.
    always @(posedge clk) begin
        if (rst) imem_addr <= 0;
        else if (pc_en) imem_addr <= redirect ? redirect_pc : imem_addr + 4;
    end
    // Flush has priority over enable: killed instructions must not survive.
    always @(posedge clk) begin
        if (rst || ifid_flush) begin
            ifid_valid <= 0;
        end else if (ifid_en) begin
            ifid_valid <= 1;
            ifid_pc <= imem_addr;
            ifid_instr <= imem_rdata;
        end
    end
    // Hold PC and IF/ID on load-use, but INSERT A BUBBLE into ID/EX.
    always @(posedge clk) begin
        if (rst || idex_flush) begin
            idex_valid <= 0;
        end else begin
            idex_valid <= ifid_valid && d_legal;
            idex_pc <= ifid_pc;
            idex_rv1 <= d_rv1; idex_rv2 <= d_rv2; idex_imm <= d_imm;
            idex_rs1 <= d_rs1; idex_rs2 <= d_rs2; idex_rd <= d_rd;
            idex_use1 <= d_use1; idex_use2 <= d_use2;
            idex_rw <= d_rw; idex_mr <= d_mr; idex_mw <= d_mw;
            idex_ai <= d_ai; idex_sub <= d_sub; idex_br <= d_br;
        end
    end
    // Older instructions continue, even while the front end is paused/flushed.
    always @(posedge clk) begin
        if (rst) exmem_valid <= 0;
        else begin
            exmem_valid <= idex_valid;
            exmem_pc <= idex_pc; exmem_result <= ex_result;
            exmem_store <= ex_rs2; exmem_rd <= idex_rd;
            exmem_rw <= idex_rw; exmem_mr <= idex_mr; exmem_mw <= idex_mw;
        end
    end
    always @(posedge clk) begin
        if (rst) memwb_valid <= 0;
        else begin
            memwb_valid <= exmem_valid;
            memwb_pc <= exmem_pc; memwb_rd <= exmem_rd;
            memwb_result <= exmem_result; memwb_load <= dmem_rdata;
            memwb_rw <= exmem_rw; memwb_mr <= exmem_mr;
        end
    end
    assign dmem_addr  = exmem_result;
    assign dmem_wdata = exmem_store;
    assign dmem_read  = !rst && exmem_valid && exmem_mr;
    assign dmem_write = !rst && exmem_valid && exmem_mw;
    assign commit_valid = !rst && memwb_valid;
    assign commit_pc = memwb_pc;
    assign commit_we = wb_we;
    assign commit_rd = memwb_rd;
    assign commit_data = wb_data;
endmodule

decoder.v
`timescale 1ns/1ps
// Teaching subset: ADD, SUB, ADDI, LW, SW, BEQ. Not a complete RV32I decoder.
module decoder (
    input  wire [31:0] instr,
    output wire [4:0]  rs1, rs2, rd,
    output reg  [31:0] imm,
    output reg uses_rs1, uses_rs2,
    output reg reg_write, mem_read, mem_write,
    output reg alu_imm, alu_sub, branch, legal
);
    assign rs1 = instr[19:15];
    assign rs2 = instr[24:20];
    assign rd  = instr[11:7];
    always @* begin
        imm = 32'b0;
        uses_rs1 = 0; uses_rs2 = 0;
        reg_write = 0; mem_read = 0; mem_write = 0;
        alu_imm = 0; alu_sub = 0; branch = 0; legal = 0;
        case (instr[6:0])
            7'b0110011: begin
                if (instr[14:12] == 3'b000 &&
                    (instr[31:25] == 7'b0000000 ||
                     instr[31:25] == 7'b0100000)) begin
                    legal = 1; uses_rs1 = 1; uses_rs2 = 1;
                    reg_write = 1; alu_sub = instr[30];
                end
            end
            7'b0010011: begin
                if (instr[14:12] == 3'b000) begin
                    legal = 1; uses_rs1 = 1; reg_write = 1; alu_imm = 1;
                    imm = {{20{instr[31]}}, instr[31:20]};
                end
            end
            7'b0000011: begin
                if (instr[14:12] == 3'b010) begin
                    legal = 1; uses_rs1 = 1; reg_write = 1;
                    mem_read = 1; alu_imm = 1;
                    imm = {{20{instr[31]}}, instr[31:20]};
                end
            end
            7'b0100011: begin
                if (instr[14:12] == 3'b010) begin
                    legal = 1; uses_rs1 = 1; uses_rs2 = 1;
                    mem_write = 1; alu_imm = 1;
                    imm = {{20{instr[31]}}, instr[31:25], instr[11:7]};
                end
            end
            7'b1100011: begin
                if (instr[14:12] == 3'b000) begin
                    legal = 1; uses_rs1 = 1; uses_rs2 = 1; branch = 1;
                    imm = {{19{instr[31]}}, instr[31], instr[7],
                           instr[30:25], instr[11:8], 1'b0};
                end
            end
            default: begin end
        endcase
    end
endmodule

regfile.v
`timescale 1ns/1ps
module regfile (
    input wire clk, rst,
    input wire [4:0] rs1, rs2, rd,
    input wire we,
    input wire [31:0] wd,
    output wire [31:0] rv1, rv2
);
    reg [31:0] regs [0:31];
    integer i;
    always @(posedge clk) begin
        if (rst) begin
            for (i = 0; i < 32; i = i + 1) regs[i] <= 0;
        end else if (we && rd != 0) begin
            regs[rd] <= wd;
        end
    end
    // Explicit WB -> ID bypass avoids reliance on simulator scheduling or
    // a particular register-file primitive's read-during-write behavior.
    assign rv1 = (rs1 == 0) ? 32'b0 :
                 (we && rd == rs1) ? wd : regs[rs1];
    assign rv2 = (rs2 == 0) ? 32'b0 :
                 (we && rd == rs2) ? wd : regs[rs2];
endmodule

forward_unit.v
`timescale 1ns/1ps
module forward_unit (
    input wire ex_valid, uses_rs1, uses_rs2,
    input wire [4:0] rs1, rs2,
    input wire m_valid, m_reg_write, m_mem_read,
    input wire [4:0] m_rd,
    input wire w_valid, w_reg_write,
    input wire [4:0] w_rd,
    output reg [1:0] forward_a, forward_b
);
    // 00: ID/EX value; 10: EX/MEM ALU result; 01: MEM/WB writeback value.
    wire m_a = ex_valid && uses_rs1 && rs1 != 0 &&
               m_valid && m_reg_write && m_rd == rs1;
    wire m_b = ex_valid && uses_rs2 && rs2 != 0 &&
               m_valid && m_reg_write && m_rd == rs2;
    wire w_a = ex_valid && uses_rs1 && rs1 != 0 &&
               w_valid && w_reg_write && w_rd == rs1;
    wire w_b = ex_valid && uses_rs2 && rs2 != 0 &&
               w_valid && w_reg_write && w_rd == rs2;
    always @* begin
        forward_a = 2'b00;
        forward_b = 2'b00;
        // The youngest matching writer owns the value. If it is a load,
        // it is not ready here; do NOT fall through to an older WB writer.
        if (m_a) begin
            if (!m_mem_read) forward_a = 2'b10;
        end else if (w_a) forward_a = 2'b01;
        if (m_b) begin
            if (!m_mem_read) forward_b = 2'b10;
        end else if (w_b) forward_b = 2'b01;
    end
endmodule

hazard_unit.v
`timescale 1ns/1ps
module hazard_unit (
    input wire d_valid, d_uses_rs1, d_uses_rs2,
    input wire [4:0] d_rs1, d_rs2,
    input wire e_valid, e_mem_read,
    input wire [4:0] e_rd,
    input wire redirect,
    output wire load_use, pc_en, ifid_en, ifid_flush, idex_flush
);
    assign load_use = d_valid && e_valid && e_mem_read && e_rd != 0 &&
                      ((d_uses_rs1 && d_rs1 == e_rd) ||
                       (d_uses_rs2 && d_rs2 == e_rd));
    assign pc_en      = redirect || !load_use;
    assign ifid_en    = !load_use;
    assign ifid_flush = redirect;
    assign idex_flush = redirect || load_use;
endmodule

tb_pipeline.sv
`timescale 1ns/1ps
// SystemVerilog is used only by the testbench; the DUT is Verilog-2001.
module tb_pipeline;
    reg clk = 0;
    reg rst = 1;
    always #5 clk = ~clk;
    wire [31:0] ia, da, dw, ir, dr, cp, cd;
    wire mr, mw, cv, cw;
    wire [4:0] cr;
    reg [31:0] imem [0:511];
    reg [31:0] dmem [0:255];
    assign ir = imem[ia[10:2]];
    assign dr = dmem[da[9:2]];
    pipeline_demo dut (
        .clk(clk), .rst(rst), .imem_addr(ia), .imem_rdata(ir),
        .dmem_addr(da), .dmem_wdata(dw), .dmem_read(mr), .dmem_write(mw),
        .dmem_rdata(dr), .commit_valid(cv), .commit_we(cw),
        .commit_pc(cp), .commit_rd(cr), .commit_data(cd)
    );
    integer cycle = 0, max_cycles, trace_fd, event_fd, i;
    reg prev_stall = 0, prev_redirect = 0;
    reg [31:0] prev_pc, prev_dpc, prev_instr, prev_target;
    reg [31:0] prev_epc;
    reg prev_evalid;
    initial begin
        for (i = 0; i < 512; i = i + 1) imem[i] = 32'h00000013;
        for (i = 0; i < 256; i = i + 1) dmem[i] = 0;
        $readmemh("program.hex", imem);
        $readmemh("memory.hex", dmem);
        if (!$value$plusargs("CYCLES=%d", max_cycles)) max_cycles = 60;
        trace_fd = $fopen("trace.csv", "w");
        event_fd = $fopen("events.csv", "w");
        $fdisplay(trace_fd, "cycle,pc_f,pc_d,pc_e,pc_m,pc_w,load_use,forward_a,forward_b,redirect,pc_en,ifid_en,ifid_flush,idex_flush,ex_a,ex_rs2,wb_we,wb_rd,wb_data,mem_we,mem_addr,mem_data");
        $fdisplay(event_fd, "kind,pc,dest,data");
        $dumpfile("wave.vcd");
        $dumpvars(0, tb_pipeline);
        repeat (2) @(negedge clk);
        #1 rst = 0;
    end
    // These samples describe the cycle ENDING at this rising edge.
    // All RHS values are sampled before DUT nonblocking assignments update.
    always @(posedge clk) begin
        if (!rst) begin
            cycle = cycle + 1;
            if (ia[1:0] != 0 || ia >= 2048) $fatal(1, "instruction address out of model");
            if ((mr || mw) && (da[1:0] != 0 || da >= 1024))
                $fatal(1, "unaligned/out-of-range data access");
            if (dut.ifid_valid && !dut.d_legal) $fatal(1, "unsupported instruction");
            if (prev_stall && !prev_redirect) begin
                if (ia !== prev_pc || dut.ifid_pc !== prev_dpc ||
                    dut.ifid_instr !== prev_instr || dut.idex_valid)
                    $fatal(1, "load-use must hold front end and bubble ID/EX");
                if (dut.exmem_valid !== prev_evalid || dut.exmem_pc !== prev_epc)
                    $fatal(1, "load must advance into MEM");
            end
            if (prev_redirect && (ia !== prev_target || dut.ifid_valid || dut.idex_valid))
                $fatal(1, "redirect must replace PC and invalidate younger stages");
            if (dut.idex_valid && dut.exmem_valid && dut.exmem_mr && dut.exmem_rd != 0 &&
                ((dut.idex_use1 && dut.idex_rs1 == dut.exmem_rd) ||
                 (dut.idex_use2 && dut.idex_rs2 == dut.exmem_rd)))
                $fatal(1, "consumer reached EX before youngest load is ready");
            if (dut.rf.regs[0] !== 0) $fatal(1, "x0 changed");
            if (mw) begin
                dmem[da[9:2]] <= dw;
                $fdisplay(event_fd,"S,%08x,%08x,%08x",dut.exmem_pc,da,dw);
            end
            if (cw) $fdisplay(event_fd,"W,%08x,%08x,%08x",cp,{27'b0,cr},cd);
            $fdisplay(trace_fd,"%0d,%08x,%08x,%08x,%08x,%08x,%0d,%0d,%0d,%0d,%0d,%0d,%0d,%0d,%08x,%08x,%0d,%0d,%08x,%0d,%08x,%08x",
                cycle,ia,dut.ifid_valid?dut.ifid_pc:32'hffffffff,
                dut.idex_valid?dut.idex_pc:32'hffffffff,
                dut.exmem_valid?dut.exmem_pc:32'hffffffff,
                dut.memwb_valid?dut.memwb_pc:32'hffffffff,
                dut.load_use,dut.forward_a,dut.forward_b,dut.redirect,
                dut.pc_en,dut.ifid_en,dut.ifid_flush,dut.idex_flush,
                dut.ex_a,dut.ex_rs2,cw,cr,cd,mw,da,dw);
            prev_stall = dut.load_use; prev_redirect = dut.redirect;
            prev_pc = ia; prev_dpc = dut.ifid_pc; prev_instr = dut.ifid_instr;
            prev_target = dut.redirect_pc;
            prev_epc = dut.idex_pc; prev_evalid = dut.idex_valid;
            if (cycle == max_cycles) begin
                $fclose(trace_fd); $fclose(event_fd);
                $display("PASS: cycle invariants (%0d cycles)", cycle);
                $finish;
            end
        end
    end
endmodule
